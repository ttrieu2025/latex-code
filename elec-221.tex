    \section{Continuous Time Signals}
    
    Even and Odd Components
     \[ x_\text{even}(t) = \frac{1}{2}[x(t) + x(-t)]\]
     \[x_\text{odd}(t) = \frac{1}{2}[x(t) - x(-t)]\]
    
    Periodicity 
    \[x(t + kT_0) = x(t)\,\forall t \in (-\infty, \infty)\]
    Energy
    \[ E = \int_{-\infty}^{\infty} \abs{x(t)}^2\,dt\]
    Power
    \[P = \lim_{T \rightarrow \infty} \frac{1}{2T} \int_{-T}^{T}\abs{x(t)}^2\,dt\]
    Causality:\\
    \begin{itemize}
    	\itemsep0em
    	\item Causal: $x(t) = 0$ for $t < 0$.
    	\item Anti-Causal: $x(t) = 0$ for $t \geq 0$.
    	\item Acausal or Non-Causal: Both of the above.
    \end{itemize}
    
    \section{Continuous Time Systems}
    \textbf{Linearity}
    \[\mathcal{S}[\alpha x(t) + \beta y(t)] = \alpha\mathcal{S}[x(t)] + \beta\mathcal{S}[y(t)]\]
    
    \textbf{Time Invariance}
    \[\mathcal{S}[x(t)] = y(t)\Rightarrow\mathcal{S}[x(t \pm \tau)] = y(t \pm \tau)\]
    
    \textbf{Zero-State Response. }Due to the input as the initial conditions are zero.\\[6pt]
    
    \textbf{Zero-Input Response. }Due to the initial conditions as the input is zero.\\[6pt]
    
    \textbf{Convolution}
    \[y(t) = \int_{-\infty}^{\infty}x(\tau)h(t-\tau)\,d\tau = \int_{-\infty}^{\infty}x(t-\tau)h(\tau)\,d\tau\]
    
    \textbf{Causality. }A continuous time system $\mathcal{S}$ is causal if whenever $x(t) = 0$ and there are no initial conditions, $y(t) = 0$ and the output $y(t)$ does not depend on future inputs.\\[6pt]
    \textbf{BIBO Stability. }If an input bounded then the output of a BIBO stable system is bounded.
    \[\Leftrightarrow \int_{-\infty}^{\infty} \abs{h(t)}\,dt < \infty\]
    
    \section{Laplace Transform}
    
    Eigenfunction Property
    \[\mathcal{S}[e^{s_0t}] = H(s_0)e^{s_0t}\]
    
    One Sided Laplace Transform:\\
    \[F(s) = \mathcal{L}[f(t)u(t)] = \int_{0^-}^{\infty}f(t)e^{-st}\,dt, \forall s = \sigma + j\omega\]\\
    
    Note: $\abs{H(s)}$ is greatest when $s$ is closest to the poles.
    
    \vspace{10pt}
    
    \begin{tabular}{ll}  
    	\textbf{Signal Support} & \textbf{ROC}\\
    	Finite support & Entire s-plane.\\
    	Causal function & $\sigma > \max(\sigma_i)$,  $-\infty < \omega < \infty$\\
    	Anti-causal & $\sigma < \min(\sigma_i)$,  $-\infty < \omega < \infty$\\
    	Non-causal & $\mathcal{R} = \mathcal{R}_\text{causal} \cap \mathcal{R}_\text{anti-causal}$\\
    \end{tabular}
    \vspace{0.5em}
    
    \textbf{Initial Value Theorem}\\
    \[ f(0^+) = \lim_{s\rightarrow\infty} sF(s)\]
    
    \textbf{Final Value Theorem}\\
    \[\lim_{t\rightarrow\infty} f(t) =\lim_{s\rightarrow 0} sF(s)\]
    
    \textbf{BIBO Stability. }If the region of convergence contains the $j\omega$-axis then the system is BIBO stable.\\[6pt]
    
    \textbf{2-Sided LT from 1-Sided LT}
    \begin{itemize}
        \item For an anti-causal function $C(s)=\mathcal{L}\left[c(-t)u(t)\right]_{(-s)}$\\
        \item For a 2-sided function $p(t) = p_{ac}(t) + p_c(t) = p(t)u(-t) + p(t)u(t)$\\
        \item 	$P(t)=\mathcal{L}\left[p_{ac}(-t)u(t)\right]_{(-s)} + \mathcal{L}\left[p_{c}(t)u(t)\right]$
    \end{itemize}	
    
    \section{CT Fourier Series}

    \[x(t)=\sum_{k=-\infty}^\infty X_ke^{jk\omega_0 t}\]
    
    Eigenfunction Property
    \[\mathcal{S}[e^{j\omega_0t}] = H(j\omega_0)e^{j\omega_0t}\]
    \begin{align*}
    x(t) &= \sum_k X_k e^{j\omega_kt} \\
    \Rightarrow y(t) &= \sum_k X_k H(j\omega_k)e^{j\omega_kt}\\
    &= \sum_k X_k \abs{H(j\omega_k)}e^{j(\omega_kt + \angle H(\omega_k))}
    \end{align*}
    
    Fourier Series Coefficients (for any $t_0$)
    \[ X_k = \frac{1}{T_0}\int_{t_0}^{t_0 + T_0}x(t)e^{-jk\omega_0t}\,dt\]
    
    Parseval's Power Relation (for any $t_0$):\\
    \[P = \frac{1}{T_0} \int_{t_0}^{t_0 + T_0}\abs{x(t)}^2\,dt = \sum_{k = -\infty}^{\infty}\abs{X_k}^2\]
    
    Symmetry of Line Spectra:\\
     $\abs{X_k} = \abs{X_{-k}}$\\
     $\angle X_k = -\angle X_{-k}$
    
    Trigonometric Fourier Series:\\
    \[x(t) = X_0 + 2\sum_{k = 1}^{\infty}\abs{X_k}\cos(k\omega t + \Theta_k)\]\\
    \[x(t) = c_0 + 2\sum_{k = 1}^{\infty}[c_k\cos(k\omega_0 t) + d_k\sin(k\omega_0 t)]\]\\
    \[c_k = \frac{1}{T_0} \int_{t_0}^{t_0 + T_0}x(t)\cos(k\omega_0 t)\,\dd t \quad k = \textbf{0},1,2,..\]\\
    \[d_k = \frac{1}{T_0} \int_{t_0}^{t_0 + T_0}x(t)\sin(k\omega_0 t)\,dt\quad k = 1,2,3,...\]\\
    \[\Theta_k = -\arctan(d_k/c_k)\]
    
    
    Fourier Coefficients from Laplace Transform. If $x_1(t)$ is a single period of $x(t)$, then
    \[X_k = \frac{1}{T_0}\mathcal{L}[x_1(t)] \big\rvert_{s = jk\omega_0}\]
    
    Response of LTI Systems to Periodic Signals. If the input to an LTI system has Fourier Series $x(t) = X_0 + 2\sum_{k=1}^{\infty}\abs{X_k}\cos(k\omega t + \angle X_k)$, then the steady state response is 
    \[ y(t) = X_0\abs{H(j0)} + 2\sum_{k=1}^{\infty}\abs{X_k}\abs{H(jk\omega_0)}\cos(k\omega_0 t + \angle X_k + \angle H(jk\omega_0))\]
    
    \section{CT Fourier Transform}
    \[X(\omega) = \int_{-\infty}^{\infty}x(t)e^{-j\omega t}\,\dd t\]\\
    Inverse Fourier Transform 
    \[x(t) = \frac{1}{2\pi}\int_{-\infty}^{\infty}X(\omega)e^{j\omega t}\,\dd\omega\]
    
    Fourier Transform from Laplace Transform (if $X(s)$ contains the $j\omega$-axis),
    \[\mathcal{F}[x(t)] = \mathcal{L}[x(t)]\big\rvert_{s = j\omega} = X(s)\big\rvert_{s = j\omega}\]
    
    Fourier Transform of Periodic Signals under Fourier Series\\
    \[ \sum_k X_k e^{jk\omega_0 t} \Leftrightarrow \sum_k 2\pi X_k \delta(\omega - k\omega_0)\]
    
    Parseval's Energy Relation
    \[ E = \int_{-\infty}^{\infty} \abs{x(t)}^2\,\dd t = \frac{1}{2\pi}\int_{-\infty}^{\infty}\abs{X(\omega)}^2\,\dd\omega\]

    \section{Sampling Theory}
     \[x_s(t) = x(nT_s) = x(t)\rvert_{t = nT_s} = x(t)\sum_n \delta(t - nT_s)\]
    \[X_s(\omega) = \frac{1}{T_s} \sum_{k=-\infty}^{\infty}X(\omega - k\omega_s)\]
    
    \textbf{Nyquist-Shannon Sampling Rate}
     \[\omega_s = \frac{2\pi}{T_s} \geq 2\omega_\text{max}\]
    \textbf{Aliasing} occurs if\,\, $\omega_s < 2\omega_\text{max}$\\[4pt]
    \textbf{Aliasing frequency mapping} for $[0, f_s/2]$
    \[f_{\text{alias}}=|f-kf_{s}|\]

    \textbf{Reconstruction} $X(\omega)=X_s(\omega)H_\text{lp}(\omega)$:
    \begin{equation*}
    \hspace{-8em}
    H_{\text{lp}}(\omega) =
    \begin{cases}
    T_s &  -\dfrac{\omega_s}{2} \leq \omega \leq \dfrac{\omega_s}{2}\\
    0 & \text{otherwise}
    \end{cases}
    \end{equation*}
    \textbf{Inclusive bounds.} If represented by a unit step function the bounds are inclusive for filters\\[6pt]
