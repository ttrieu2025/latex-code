    \subsection*{Fundamental Set of Solutions}
    Wronskian Matrix
    \[
    W[y_1,y_2](t) =\det
    \begin{bmatrix}
    y_1(t) & y_2(t) \\
    y_1'(t) & y_2'(t)
    \end{bmatrix}\]
    $W(t)\ne 0 \,\, \text{for some } t_0 \in I\Rightarrow y_1, y_2$ form a fundamental set of solutions. 

    \subsection*{1. First-Order Linear ODE}
        \textbf{Constant-coefficient}\qquad$y'+ay=0$\\
    General solution \qquad \,\,\,\qquad $y(t) = Ce^{-at}$\\
    Inhomogeneous\qquad \qquad\,\,\,\,\, $y'(t) + p(t)y(t) = g(t)$\\
    Integrating factor \[ \displaystyle y(t) = \frac{1}{\mu(t)}\left[ \int^t \mu(s)g(s) \dd s+C\right ], \mu(t) = e^{\int^tp(s)\dd s} \]

    \subsection*{2. Second-Order Linear ODE}
        \textbf{Constant-coefficient}\qquad $ay''+by'+ cy=0$\\
        Characteristic Equation \qquad $ar^2+br+c=0$\\
        General Solution
        \begin{align*}
        &\Delta > 0:\,r_1 < r_2 && y(t) = c_1 e^{r_1 t} + c_2 e^{r_2 t} \\[1pt]
        &\Delta = 0:\, r_1 = r_2 && y(t) = (c_1 + c_2 t)e^{r t}\\[1pt]
        &\Delta < 0:\,r = \lambda \pm \mu i && 
        y(t) = e^{\lambda t}\big(c_1 \cos(\mu t) + c_2 \sin(\mu t)\big)
        \end{align*}

        \textbf{Euler's Equation} \qquad \qquad $at^2y''+bty'+cy=0$\\
        Characteristic equation \qquad $ak(k-1)+bk+c=0$\\
        General Solution
        \begin{align*}
        &\Delta > 0:\, k_1<k_2 &&y(t) = c_1t^{k_1}+c_2 t^{k_2}\\
        &\Delta = 0:\, k_1=k_2\, &&y(t) = t^k(c_1+c_2\ln t )\\
        &\Delta < 0:\, k= \alpha \,\pm\, \beta i &&y(t) = t^\alpha ( c_1 \cos(\beta \ln t)  + c_2\sin(\beta \ln t))
        \end{align*}

    \subsubsection{Reduction of Orders} 
        To find a general solution of a second-order, linear ODE 
        \[y'' +p(t)y' +q(t)y = g(t)\]
        1. Find one particular solution $y_1$ of the ODE \\
        \[\displaystyle y_1'' +p(t)y_1' +q(t)y_1 = 0\]
        2. Find a solution $y$ in the form of \[y(t) = u(t)y_1(t)\]\\
        3. Compute $y'=uy_1'+u'y_1$ and $y''= uy_1''+2u'y'_1+u''y_1$\\
        4. Substitute in $y''+py'+qy= (2y_1'+py_1)u' + (y_1)u''$. After substitution, the $u$ terms cancel.\\
        5. Let $A(t) = u'$ then the equation becomes a $1^\text{st}$ linear ODE \[\displaystyle A'(t) + \left(2\frac{y_1'}{y_1}+p\right)A(t) = \frac{g}{y_1}\]\\
        6. Solve by integrating factor \[\displaystyle \mu(t) = \exp\left( \int^t \left( 2\frac{y_1'(s)}{y_1(s)} + p(s) \right) ds \right)\]\\
        7. General solution \[\displaystyle y(t) = c_1 y_1(t) + y_1(t) \left( \int^t \left[ \frac{1}{\mu(s)} 
            \int^s \mu(\tau) \frac{g(\tau)}{y_1(\tau)} \, \dd\tau \right] \dd s \right)\]

    \subsubsection{Method of Variation of Parameter}
    Constant-coefficient\qquad $\mathbf{x}' = A\mathbf{x} + \mathbf{f}(t)$\\

    \begin{enumerate}
    \item Find a fundamental matrix
    \[\Phi(t) = \big[ \mathbf{x}_1(t)\ \mathbf{x}_2(t)\ \cdots\ \mathbf{x}_n(t) \big].\]

    \item Compute the inverse matrix $\Phi^{-1}(t)$.

    \item Form the integrand $\displaystyle \Phi^{-1}(t)\mathbf{f}(t)$.

    \item Integrate $ \displaystyle \int_{t_0}^{t} \Phi^{-1}(s)\mathbf{f}(s)\,\dd s.$

    \item Multiply by $\Phi(t)$
    \[\mathbf{x}_p(t) = \Phi(t)\int_{t_0}^{t} \Phi^{-1}(s)\mathbf{f}(s)\,ds.
    \]

    \item Write the general solution
    \[
    \mathbf{x}(t) = \mathbf{x}_p(t) + \Phi(t)\mathbf{c}.
    \]

    \item To solve an IVP,  If $\mathbf{x}(t_0)=\mathbf{x}_0$, choose
    \[
    \mathbf{c} = \Phi^{-1}(t_0)\mathbf{x}_0.
    \]
    \end{enumerate}
    Ex: Solve the system

    \mathbf{x}' =
    \begin{bmatrix}
    -2 & 1 \\
    1 & -2
    \end{bmatrix}
    \mathbf{x}

    \subsubsection{Eigenvalues and Eigenfunctions}
        Equation $y'' + \lambda y = 0$ \\[6pt]
        \textbf{Dirichlet BCs} $y(0) = 0 = y(L)$ \\[6pt]
        Eigenvalues \quad \,\,\,$\displaystyle \lambda_n = \left(\frac{n\pi}{L}\right)^2, n = 1,2,3,...$    \\[6pt]
        Eigenfunctions \,\,$\displaystyle y_n(x) = \sin\left(\frac{n\pi}{L}x\right) $    \\[6pt]
        \textbf{Neumann BCs} $y'(0) = 0 = y'(L)$ \\[6pt]
        Eigenvalues \quad \,\, \,$\displaystyle \lambda_n = \left(\frac{n\pi}{L}\right)^2, n = \textbf{0},1,2,3,...$    \\[6pt]
        Eigenfunctions \,\,$\displaystyle y_n(x) = \cos\left(\frac{n\pi}{L}x\right) $\\[6pt]
        \textbf{Orthogonality of eigenfunctions}\\
        We have two ODEs $(B_1)$ and $(B_2)$ respectively
        \begin{align*}
        &(B_1)\, y'' + \lambda y=0,&y(a)=0 =y(b)\\
        &(B_2)\, y'' + \lambda y=0,&y'(a) = 0=y'(b)
        \end{align*}
        Let $y_1$ and $y_2$ be eigenfunctions of $(B_1)$ (or $(B_2)$ with eigenvalues $\lambda_1\ne\lambda_2$ then
        \[\int_a^by_1(x)y_2(x)\dd x=0.\]
        \textbf{Consequences}
        \begin{enumerate}
            \item $\displaystyle \int_{0}^{L}
    \sin\!\left(\frac{m\pi x}{L}\right)
    \sin\!\left(\frac{n\pi x}{L}\right)\,dx
    =
    \begin{cases}
    0, & m\neq n,\\[6pt]
    \dfrac{L}{2}, & m=n.
    \end{cases}$
            \item $\displaystyle \int_{0}^{L}
    \cos\!\left(\frac{m\pi x}{L}\right)
    \cos\!\left(\frac{n\pi x}{L}\right)\,dx
    =
    \begin{cases}
    0, & m\neq n,\\
    L, & m=n=0,\\
    \dfrac{L}{2}, & m=n\neq 0.
    \end{cases}
    $
            \item $\displaystyle \int_{0}^{L}
    \sin\!\left(\frac{m\pi x}{L}\right)
    \cos\!\left(\frac{n\pi x}{L}\right)\,dx
    =
    0
    \quad \forall\, m,n.$
        \end{enumerate}

    \subsection{4. Partial Differential Equations}
        
        \subsubsection{Heat Equation}
        $\begin{cases} u_t = \alpha^2 u_{xx}, \quad 0 < x < L, \quad t > 0\\
        u(x,0) = f(x)\\
        \end{cases}$ \\[6pt]
        Separation form \qquad $u(x,t) = X(x)T(t)$
        \[\Rightarrow \frac{X''(x)}{ X(x)}=\frac{T'(t)}{\alpha T(t)} = -\lambda\]
        \textbf{Dirichlet BCs}\,\textbf{[}$u(0,t) = 0 = u(L,t)$\textbf{]}  \\
        \,\,$\displaystyle u(x,t) =\sum_{n=1}^\infty b_n e^{-(n \pi \alpha/L)^2 t} \sin\left(\frac{n\pi x}{L}\right) $\\[6pt]
        \qquad$\displaystyle b_n = \frac{2}{L}\int_0^L f(x)\sin\left(\frac{n\pi x}{L}\right)\, \dd x$\\[10pt]    
        \textbf{Neumann BCs}\,\textbf{[}$u_x(0,t) = 0 = u_x(L,t)$\textbf{]} \\
        \,\,$\displaystyle  u(x,t) = \frac{c_0}{2} + \sum_{n=1}^\infty c_n e^{-(n \pi \alpha/L)^2 t} \cos\left(\frac{n\pi x}{L}\right)$\\[6pt]
        \qquad $\displaystyle c_0 = \frac{2}{L}\int_0^Lf(x)\, \dd x$
        \qquad $\displaystyle \displaystyle c_n = \frac{2}{L} \int_0^L f(x)\cos\left(\frac{n\pi x}{L}\right) \mathrm{d} x$\\[10pt]
        Inhomogeneous\qquad $u_t = \alpha^2 u_{xx} + f(x)$.

        
        \textbf{Step 1.} Find steady-state function $u_\text{ss}$. At a steady state, $u_{t} =0$, so $\displaystyle u_{xx} = -f(x,t)/\alpha^2$\\
        \textbf{Step 2.} $u(x,t) = u_\text{ss} + v(x,t)$ with $v(x,t)$ is the solution of the homogeneous heat equation.\\[6pt]
        \subsubsection{Wave Equation}
        $\begin{cases}y_{tt} = c^2 y_{xx}, \quad 0 < x < L, \quad t > 0\\
        y(x,0) = f(x)\\
        y_t(x,0) = g(x)\\
        \end{cases}$\\[10pt]
        Separation form \qquad $y(x,t) = X(x)T(t)$\\
